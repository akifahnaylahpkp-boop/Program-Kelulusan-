    <script>
        function cekKelulusan() {

            let nilai = Number(document.getElementById("nilai").value);
            let kehadiran = Number(document.getElementById("kehadiran").value);
            let hasil = document.getElementById("hasil");

            // Nested If
            if (nilai >= 75) {

                if (kehadiran >= 80) {
                    hasil.innerHTML = "🎉 SELAMAT! ANDA LULUS 🎉";
                    hasil.className = "lulus";
                } else {
                    hasil.innerHTML = "❌ TIDAK LULUS<br>Kehadiran kurang dari 80%.";
                    hasil.className = "tidak-lulus";
                }

            } else if (nilai >= 70) {

                if (kehadiran > 90) {
                    hasil.innerHTML = "🎉 SELAMAT! ANDA LULUS 🎉";
                    hasil.className = "lulus";
                } else {
                    hasil.innerHTML = "❌ TIDAK LULUS";
                    hasil.className = "tidak-lulus";
                }

            } else {

                hasil.innerHTML = "❌ TIDAK LULUS<br>Nilai di bawah 70.";
                hasil.className = "tidak-lulus";
            }

            hasil.style.display = "block";
        }
    </script>
